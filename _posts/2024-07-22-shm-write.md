#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/mman.h>
#include <fcntl.h>
#include <semaphore.h>
//ftruncate
#include <unistd.h>

// common for two processes
#define SHM_NAME "/shared_memory"
#define SEM_NAME "/mysemaphore"
#define PERSON_SIZE 10

typedef struct {
    char FirstName[20];
    char LastName[20];
} Person;

int main(){
	// declare a shm handle
	int fd;
	int status;
	sem_t *sem; // declare a semaphore for mutex lock
	void *addr;
	Person *SharedPerson;

	// open the semaphore
	
	sem = sem_open(SEM_NAME, O_CREAT, S_IRUSR | S_IWUSR, 1);
	if (sem == SEM_FAILED) {
		perror("sem_open");
		exit(-1);
	}

	// open the shared memory
	fd = shm_open(SHM_NAME, O_CREAT | O_RDWR, S_IRUSR | S_IWUSR);
	if (fd == -1) {
			perror("shm_open");
			exit(-1);
	}
	status = ftruncate(fd, sizeof(Person)* PERSON_SIZE);

	if (status == -1) {
		perror("ftruncate");
		exit(-1);
	}

	addr = mmap(NULL, sizeof(Person) * PERSON_SIZE, PROT_READ|PROT_WRITE, MAP_SHARED, fd, 0);
	if (addr == MAP_FAILED) {
		printf("Map failed\n");
		exit(-1);
	}
	//close fd
	status = close(fd);
	if (status == -1) {
		perror("close");
		exit(-1);
	}
	// aquire lock to write to shared memory
	if (sem_wait(sem) == -1) {
		perror("sem_wait");
		exit(-1);
	}
	printf("Writing to shared memory\n");
	SharedPerson = (Person *)addr;
	
	for (int i = 0; i < 10; i++) {
		snprintf(SharedPerson[i].FirstName, 20, "First Name %d", i);
		snprintf(SharedPerson[i].LastName, 20, "Last Name %d", i);
	}

	// release lock
	if (sem_post(sem) == -1) {
		perror("sem_post");
		exit(-1);
	}
	
	status = sem_close(sem);
	if (status == -1) {
		perror("sem_close");
		exit(-1);
	}	
	/*
	if (munmap(addr, sizeof(Person)) == -1) {
		printf("Unmap failed\n");
		exit(-1);
	}*/
	
}
